create table Fecha (
  id int,

);

/* fecha inicial y la fecha final (la fecha actual) */
%let fecha_inicial = '31JAN2017'd;
%let fecha_final = %sysfunc(today());

/* Utiliza un bucle para insertar el último día de cada dia */
data Fecha;
  format fecha DDMMYY10.;
  do id = 1 to intck('day', &fecha_inicial, &fecha_final);

  end;

/* Cantidad de AP que pasaron a la firma del subsecretario con la fecha minima */ 

proc sql;
create table Work.LogAPfirmaminimascretario AS
select distinct
    t1.averiguacionPrevia,
    min(t1.fechaHora) as MinimaFecha
from
    SIGAD.LogEstado t1
left join
    SIGAD.EstadoDelTramiteAP t2 on (t2.id = t1.nuevoEstadoAP_id)
where
    t2.nombre='A LA FIRMA DEL SUBSECRETARIO' and t1.averiguacionPrevia IS NOT NULL
group by
    t1.averiguacionPrevia;
quit;

/*otra forma te obtener la fecha max para sumario */

proc sql;
create table FechaMaximaLog as
select
    numerolog,
    MaximaFecha
from
    (select
        cats(c.denuncia, c.sumario, c.averiguacionPrevia) as numerolog,
        max(c.fechaHora) as MaximaFecha
    from
        SIGAD.LogEstado c
    group by denuncia, sumario, averiguacionPrevia);
quit;



proc sql;
create table SumarioFecha as
select
t1.numero,
cats(t1.numero, t2.lp) as NumeroMasLp,
t5.nombre as tipodesancion,
t2.fechaSancion,
t5.nombre as SumarioSancion,
t1.fechaInicioSumario as FechaInicio,
t1.fechaAceptacionDeCompetencia,
t1.fechaIncompetencia,
t1.fechaIngresoOTCE,
t1.fechaFinSumario,
t1.fechaDelHecho,
t2.fechaPRERESOsancion,
t4.MaximaFecha as UltimaFechaEstado,
t3.nombre as EstadoActuacion
from SIGAD.Sumarios t1
left join SIGAD.PersonaSumariada as t2 on(t2.sumario_numero=t1.numero)
left join SIGAD.EstadoDelTramiteSumario as t3 on(t1.idEstadoDelTramiteSumario=t3.id)
left join FechaMaximaLog t4 on (t4.numerolog = t1.numero)
left join SIGAD.SancionSumario as t5 on(t2.sancion=t5.id);
quit;


/* denuncias*/


proc sql ;
create table FechaDenuncias as
select
t2.numero,
t2.fechaintervencionOTCE,
t2.fechaFin,
t2.denuncianteLp,
t2.fechaDelHecho,
t5.nombre as estado,
t3.MaximaFecha as UltimaFechaEstado
from SIGAD.Denuncia t2
left join FechaMaximaLog t3 on (t3.numerolog = t2.numero)
left join SIGAD.EstadoDelTramiteDenuncia  t4 on (t2.idEstadoDelTramiteDenuncia=t4.id)
left join SIGAD.GrupoEstadoDenuncia  t5 on (t4.grupo=t5.id);
quit;

/* Averiguacion Previa*/

proc sql;
create table AveriguacionPrevia as 
select 
    t3.numero,
    t3.fechaFin,
    t3.fechaInicio,
    t4.fechaHora,
	t7.MaximaFecha as UltimaFechaEstado,
	t9.nombre as Estado,
    case when min(DATEPART(t4.fechaHora))>'11JUL2019'd then  min(DATEPART(t4.fechaHora)) else DATEPART(t3.fechaInicio) end  format=date9. as APMas,
	(case when t10.MinimaFecha is not null then DATEPART(t10.MinimaFecha)
		 when t3.fechaFin is not null then DATEPART(t3.fechaFin)
else 0 end)format=date9. as  FechaFinOFechaFirma
from SIGAD.AveriguacionPrevia t3
left join SIGAD.LogEstado as t4 on (t3.numero = t4.averiguacionPrevia)
left join FechaMaximaLog as t7 on(t3.numero=t7.numerolog)
left join SIGAD.EstadoDelTramiteAP as t8 on(t3.estadoDelTramiteAP = t8.id)
left join SIGAD.GrupoEstadoAP as t9 on(t8.grupo=t9.id)
left join  Work.LogAPfirmaminimascretario as t10 on (t3.numero=t10.averiguacionPrevia)
group by t3.numero, t3.fechaFin, t3.fechaInicio;
quit;

proc sql;
create table SuParaAceptar as
select 
  fecha as Fecha,
  case 
    when count(distinct s.numero) > 0 then count(distinct s.numero) 
    else 0 
  end as SuParaAceptar
from
  Sumariofecha s
where  
 ( DATEPART(s.fechaIngresoOTCE) <= (fecha) and DATEPART(s.fechaAceptacionDeCompetencia)>fecha )
group by fecha;
quit;

proc sql;
create table SuParaAceptar2 as
select 
  f.fecha,
  case when count(distinct s.numero) > 0 then count(distinct s.numero) else 0 end  as Actuaciones_Iniciadas
from
  Fecha f
left join
  SumarioFecha s
on
  DATEPART(s.fechaIngresoOTCE) <= f.fecha 
where
  s.fechaAceptacionDeCompetencia is null  and s.fechaIngresoOTCE is not null 
group by
  f.fecha;
quit;

proc sql;
create table SuParaAceptar3 as
select 
  f.fecha,
  case when count(distinct s.numero) > 0 then count(distinct s.numero) else 0 end  as Actuaciones_Iniciadas2
from
  Fecha f
left join
  SumarioFecha s
on
  ( DATEPART(s.fechaIngresoOTCE) <= (f.fecha) and DATEPART(s.fechaAceptacionDeCompetencia)>f.fecha )
group by
  f.fecha;
quit;

proc sql;
create table tabla1S as
select 
	f.fecha,
	t1.Actuaciones_Iniciadas as data1,
	t2.Actuaciones_Iniciadas2 as data2,
	t3.SuParaAceptar as data3,
	t2.Actuaciones_Iniciadas2 + t1.Actuaciones_Iniciadas 
	-case when t3.SuParaAceptar is not null then sum(t3.SuParaAceptar) else 0 end  as SuParaAceptar
	from
  fecha f
left join SuParaAceptar2 t1 on  (t1.fecha=f.fecha)
left join SuParaAceptar3 t2 on  (t2.fecha=f.fecha)
left join SuParaAceptar t3 on  (t3.fecha =f.fecha)
group by
f.fecha;
quit;

/*Cantidad de DN con fecha intervención OTCE en el dia campo Fecha y con el campo LP del denunciante no vacío o <> a "No Aporta"*/


proc sql;
create table tabla18C as
select 
	f.fecha,
SUM(CASE 
    WHEN DATEPART(s.fechaintervencionOTCE) IS NOT NULL 
        AND DATEPART(s.fechaintervencionOTCE) = f.fecha 
        AND s.denuncianteLp IS NOT NULL 
        AND s.denuncianteLp <> 'No aporta'
        AND MISSING(INPUT(s.denuncianteLp, BEST.)) = 0
    THEN 1 
    ELSE 0 
END) as Confianza_Q,
	sum(
    case 
        when DATEPART(s.fechaintervencionOTCE)= f.fecha
        and year(DATEPART(s.fechaintervencionOTCE))= YEAR(f.fecha) 
        then 1 
        else 0 
	end
) as Confianzapor,
	SUM(CASE 
    WHEN DATEPART(s.fechaintervencionOTCE) IS NOT NULL 
        AND DATEPART(s.fechaintervencionOTCE) = f.fecha 
        AND s.denuncianteLp IS NOT NULL 
        AND s.denuncianteLp <> 'No aporta'
         AND MISSING(INPUT(s.denuncianteLp, BEST.)) = 0
    THEN 1 
    ELSE 0 
END) / sum(
    case 
        when DATEPART(s.fechaintervencionOTCE)= f.fecha 
        then 1 
        else 0 
    end
) as Resultado
from
  fecha f
left join
  FechaDenuncias s
on
  DATEPART(s.fechaintervencionOTCE) = f.fecha 
group by
  f.fecha;
quit;

  SumarioFecha s
on
  DATEPART(s.fechaIncompetencia) <= f.fecha 
where 
s.EstadoActuacion = '12) Incompetencia' 
group by
  f.fecha;
quit;




proc sql;
create table auxiliar_SUStock as
select 
  t1.fecha,
  t2.dato1su,
  t3.dato2su,
  t4.dato3su,
  coalesce(t2.dato1su,0) - coalesce(t3.dato2su, 0) - coalesce(t4.dato3su,0) as SUStock_Dias
   from fecha t1
   left join auxiliar_SUSstock0 t2 on (t1.fecha = t2.fecha)
   left join auxiliar_SUSstock1 t3 on (t1.fecha = t3.fecha)
   left join auxiliar_SUSstock2 t4 on (t1.fecha = t4.fecha);
quit;

/* Tabla auxiliar para DNStock_90 dias */
proc sql;
create table auxiliar_DNStock as
select 
  t0.fecha,
  (select SUM(COALESCE(t7.DNFechaIntervencionOTC, 0) - COALESCE(t8.DNFechaFinMenos, 0)) 
   from fecha t1
   left join tabla7D t7 on (t1.fecha = t7.fecha)
   left join tabla9D t8 on (t1.fecha = t8.fecha)
   where t1.fecha <= t0.fecha) as DNStock_Dias
from fecha t0;
quit;

/* Tabla auxiliar para APStock_90 dias */
proc sql;
create table auxiliar_APStock as
select 
  t0.fecha,
  (select SUM(COALESCE(t9.APFechaInicioOFechaHora, 0) - COALESCE(t10.APFechaFinMenos, 0)) 
   from fecha t1
   left join tabla10A t9 on (t1.fecha = t9.fecha)
   left join tabla11A t10 on (t1.fecha = t10.fecha)
   where t1.fecha <= t0.fecha) as APStock_Dias
from fecha t0;
quit;

/*Auxiliar para calcula tabla FIRMA O FECHA FIN */

proc sql;
create table AuxiliarFirmaSubOFechaFin as
select 
  f.fecha,
   case when count(distinct s.numero) > 0 then count(distinct s.numero) else 0 end as CantidadFirmaOfechafin
from
  fecha f
left join
  AveriguacionPrevia as s on (s.FechaFinOFechaFirma = f.fecha) 
group by
  f.fecha;
quit;


/* Tabla auxiliar para APStock_90 dias _Firma_Secretaria O Fecha Fin */
proc sql;
create table FechaDeFirmaSubOFechaFin as
select 
  t0.fecha,
  (select 
	SUM(COALESCE(t9.APFechaInicioOFechaHora, 0) - COALESCE(t10.CantidadFirmaOfechafin, 0)) 
   from fecha t1
   left join tabla10A t9 on (t1.fecha = t9.fecha)
   left join AuxiliarFirmaSubOFechaFin t10 on (t1.fecha = t10.fecha)
   where t1.fecha <= t0.fecha) as APStock_Firma_o_FechaFin
from fecha t0;
quit;
